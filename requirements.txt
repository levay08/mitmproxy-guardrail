# What this guardrail needs. Two sections in one file.
#
# 1. System tools: checked first, and only what is missing gets installed, using the package
#    name of the family the installer detected.
#      name | how to check it | alpine | debian | fedora | suse | arch
#      command:<name>   the command is on PATH
#      module:<name>    python3 can import it
#      run:<command>    the command exits 0
#
bash    | command:bash               | bash      | bash            | bash        | bash        | bash
python3 | command:python3            | python3   | python3         | python3     | python3     | python
venv    | module:ensurepip                | python3   | python3-venv    | python3     | python3     | python
pip     | run:python3 -m pip --version | py3-pip | python3-pip     | python3-pip | python3-pip | python-pip
openssl | command:openssl            | openssl   | openssl         | openssl     | openssl     | openssl
curl    | command:curl               | curl      | curl            | curl        | curl        | curl
nft     | command:nft                | nftables  | nftables        | nftables    | nftables    | nftables
sudo    | command:sudo               | sudo      | sudo            | sudo        | sudo        | sudo
ss      | command:ss                 | iproute2  | iproute2        | iproute     | iproute2    | iproute2
pgrep   | command:pgrep              | procps    | procps          | procps-ng   | procps      | procps-ng
certutil| command:certutil           | nss-tools | libnss3-tools   | nss-tools   | mozilla-nss-tools | nss
